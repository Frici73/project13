-- Adattáblák

// Személy mint felhasználó
ProfilPerson
    INT person_id Auto_Inc - Primary key
    VARCHAR(100) p_email - NOT NULL, UNIQUE
    VARCHAR(40) p_username - NOT NULL
    VARCHAR(255) p_password - NOT NULL // hash-elni kell
    VARCHAR(56) p_country - NOT NULL
    VARCHAR(200) p_location - NOT NULL // község
    INT p_fake_datas - NOT NULL, DEFAULT 0, CHECK >= 0
    MEDIUMBLOB p_image_profil
    BOOLEAN driving_license - NOT NULL, DEFAULT FALSE
    VARCHAR(20) p_phone_number - NOT NULL, UNIQUE
    INT wanted_salary - NOT NULL, CHECK >= 0
    // Bérigény, havi bruttó Ft

// Képesítések foka és típusa
Qualification
    INT qualification_id Auto_Inc - Primary key
    VARCHAR(50) degree - NOT NULL
    VARCHAR(50) type - NOT NULL
    UNIQUE (degree, type)

// Cég mint felhasználó
ProfilCompany
    INT company_id Auto_Inc - Primary key
    VARCHAR(100) c_email - NOT NULL, UNIQUE
    VARCHAR(40) c_username - NOT NULL
    VARCHAR(255) c_password - NOT NULL // hash-elni kell
    VARCHAR(56) c_country - NOT NULL
    INT c_fake_datas - NOT NULL, DEFAULT 0, CHECK >= 0
    VARCHAR(20) c_phone_number - NOT NULL, UNIQUE
    MEDIUMBLOB c_image_profil
    VARCHAR(11) tax_id - NOT NULL, UNIQUE

// Cégek által közzétett hirdetések
AdvertisementFromCompany
    INT advertisement_id Auto_Inc - Primary key
    VARCHAR(200) c_location - NOT NULL // község, utca, házszám
    INT offered_salary - NOT NULL, CHECK >= 0
    // Felajánlott fizetés, havi bruttó Ft
    VARCHAR(100) offered_job - NOT NULL // munka neve
    TEXT requirement - NOT NULL
    INT company_id - NOT NULL, Foreign key

    Foreign key:
        company_id -> ProfilCompany.company_id
        ON DELETE CASCADE

-- Kapcsolatok

// Álláskereső és képesítés kapcsolata (N:M)
HasQualification
    INT person_id - NOT NULL, Foreign key
    INT qualification_id - NOT NULL, Foreign key

    Összetett Primary key:
        person_id & qualification_id

    Foreign keys:
        person_id -> ProfilPerson.person_id
        qualification_id -> Qualification.qualification_id

    ON DELETE CASCADE mindkét idegen kulcsnál

// Jelentkezés az álláshirdetésre (N:M)
Applying
    INT person_id - NOT NULL, Foreign key
    INT advertisement_id - NOT NULL, Foreign key
    MEDIUMBLOB cv // önéletrajz

    Összetett Primary key:
        person_id & advertisement_id

    Foreign keys:
        person_id -> ProfilPerson.person_id
        advertisement_id -> AdvertisementFromCompany.advertisement_id

    ON DELETE CASCADE mindkét idegen kulcsnál
