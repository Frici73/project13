-- adattáblák

// személy mint felhasználó
ProfilPerson
    int personid Auto_Inc - Primary key
    Varchar(30) p_email
    Varchar(40) p_username
    Varchar(255) p_password // titkosítani kell
	VARCHAR(56) p_country
	VARCHAR(200) p_location // község, utca, házszám
	int p_fakeDatas
	blob p_imageprofil
	bool driving_license
	int p_phonenumber
	int wanted_salary // bérigény

// képesítések foka és típusa
Qualification
    qualificationid - Primary key
    VARCHAR(50) degree
    VARCHAR(50) type

// cég mint felhasználó
ProfilCompany
	int companyid Auto_Inc - Primary key
    Varchar(30) c_email
    Varchar(40) c_username
    Varchar(255) c_password // titkosítani kell
	VARCHAR(56) c_country
	int c_fakeDatas
	int c_phonenumber
	blob c_imageprofil
	Varchar(11) taxID

// cégek által kitett hirdetések
AdvertisementFromCompany
    int advertisementid Auto_Inc - Primary key
	VARCHAR(200) c_location // község, utca, házszám
    int offered_salary 		// fel ajánlott fizetés
    Text requirement
	ProfilCompany.companyid - foreign key

-- kapcsolatok

// állás kereső és a képesítés kapcsolata
JobQualification
    ProfilPerson.personid & Qualification.qualificationid - Primary key

// jelentkezés a hirdetésre
Applying
	ProfilPerson.personid & AdvertisementFromCompany.advertisementid - Primary key
    blob cv
