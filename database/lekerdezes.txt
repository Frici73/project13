Group 1 // alap adatok
SELECT p_email, p_username, p_country, driving_license, p_phone_number
FROM ProfilPesron;

Group 2 // cég alap adatai
SELECT c_email, c_username, c_country, c_phone_number
FROM ProfileCompany;

Group 3 végzettség adatai (személy szerint)
SELECT p.p_username, q.type, q.degree
FROM Qualification as q
INNER JOIN PersonProfil as p
ON p.person_id = q.person_id
GROUP BY p.person_id;

Group 4 // statisztikai adatok (hol mennyiért dolgoznál)
SELECT wanted_salary, p_country, p_location
FROM ProfilPerson;

Group 5 // állás hirdetés adatai

megjegyzés: offered_job-ot be kell tenni a modellekbe

SELECT offered_job, offered_salary, requirement, c_location
FROM AdvertisementFromCompany;

Group 6 // Állásra jelentkezés adatai
SELECT a.cv, p.p_username, p.driving_license, p.p_email
FROM Applying as a
INNER JOIN ProfilPerson as p
ON p.person_id = a.person_id;
